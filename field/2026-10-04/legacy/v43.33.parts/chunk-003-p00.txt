}
function glaireCleanNoteNode(node) {
    if(!node) return '';
    const clone=node.cloneNode(true);
    clone.querySelectorAll?.('.mw-cite-backlink,.mw-cite-backlink a,.reference-accessdate,.noprint').forEach(x=>x.remove());

}
function glaireParseBookHtml(html, reference) {
    if(typeof DOMParser==='undefined') throw new Error('DOMParser unavailable in this runtime.');
    const doc=new DOMParser().parseFromString(String(html||''),'text/html');
    const root=doc.querySelector('.mw-parser-output') || doc.body;
    const kids=[...root.children];
    let start=-1,end=kids.length;
    for(let i=0;i<kids.length;i++){
        const n=glaireChapterNumber(kids[i].textContent);
        if(n===reference.chapter){ start=i; break; }
    }
    if(start<0) throw new Error(`Glaire–Vigouroux chapter ${reference.chapter} not found.`);
    for(let i=start+1;i<kids.length;i++){
        if(glaireChapterNumber(kids[i].textContent)!=null){ end=i; break; }
    }
    let verse=null,lineStart=true,seenVerse=false;
    const noteIdsByVerse=new Map(), overview=[];
    const addNote=(v,id)=>{
        if(!Number.isInteger(v)||v<reference.start||v>reference.end||!id) return;
        if(!noteIdsByVerse.has(v)) noteIdsByVerse.set(v,[]);
        const arr=noteIdsByVerse.get(v); if(!arr.includes(id)) arr.push(id);
    };
    const walk=(node)=>{
        if(node.nodeType===3){
            const txt=String(node.nodeValue||'');
            const parts=txt.split(/(\n)/);
            for(const part of parts){
                if(part==='\n'){ lineStart=true; continue; }
                if(lineStart){
                    const m=part.match(/^\s*(\d{1,3})\.\s*/);
                    if(m){ verse=+m[1]; seenVerse=true; lineStart=false; }
                    else if(part.trim()) lineStart=false;
                }
            }
            return;
        }
        if(node.nodeType!==1) return;
        const el=node;
        const tag=el.tagName;
        if(tag==='BR'){ lineStart=true; return; }
        if(tag==='SUP' && el.classList?.contains('reference')){
            const a=el.querySelector('a[href^="#cite_note"]');
            if(a) addNote(verse, decodeURIComponent(a.getAttribute('href').slice(1)));
            return;
        }
        const block=/^(P|DIV|LI|BLOCKQUOTE|DD|DT)$/.test(tag);
        if(block) lineStart=true;
        for(const child of [...el.childNodes]) walk(child);
        if(block) lineStart=true;
    };
    for(let i=start+1;i<end;i++){
        const el=kids[i];
        if(!seenVerse){
            const txt=String(el.textContent||'').replace(/\s+/g,' ').trim();
            if(txt && !/^\d{1,3}\.\s/.test(txt) && txt.length<1000 && !/^(modifier|Image:|Séparateur)$/i.test(txt)) overview.push(txt);
        }
        walk(el);
    }
    const entries=[];
    for(const [v,ids] of [...noteIdsByVerse.entries()].sort((a,b)=>a[0]-b[0])){
        let n=0;
        for(const id of ids){
            const li=doc.getElementById(id), body=li?.querySelector('.reference-text') || li;
            const text=glaireCleanNoteNode(body);
            if(!text) continue; n++;
            entries.push({label:`v. ${v}${ids.length>1?' · note '+n:''}`,start:v,end:v,html:htmlEscapeText(text)});
        }
    }
    const chapterSummary=overview.slice(0,2).join(' ').replace(/\s+/g,' ').trim();
    return {entries,chapterSummary};
}
async function loadGlaireVigourouxFr(fetcher, reference) {
    const title=glaireVigourouxTitle(reference);

    if(data?.error) throw new Error(data.error.info || data.error.code || 'Wikisource parse error.');
    const html=typeof data?.parse?.text==='string'?data.parse.text:data?.parse?.text?.['*'];
    if(!html) throw new Error('Wikisource returned no Glaire–Vigouroux transcription.');
    const parsed=glaireParseBookHtml(html,reference), sourceUrl=glaireVigourouxUrl(title);

}

        }
        if (target.dataset.date) {
            this.changeDate(target.dataset.date);
            return;
        }
        if (target.dataset.formulary !== undefined) {
            const index = Number(target.dataset.formulary);
            if (Number.isInteger(index) && index >= 0) {
                this.resolver.formularies.set(this.store.getState().selectedDate, index);
                void this.resolve(this.store.getState().selectedDate);
            }
            return;
        }
        if (target.dataset.resumeMass !== undefined) {
            void this.resumeSavedMass();
            return;
        }
        if (target.dataset.homeSheetClose !== undefined) {
            this.store.dispatch({ type: 'home-sheet', sheet: null });
            return;
        }
        if (target.dataset.homeOpenSettings !== undefined) {
            this.store.dispatch({ type: 'home-sheet', sheet: 'settings' });
            return;
        }
        if (target.dataset.settingForm) {
            this.store.dispatch({ type: 'live-form', form: target.dataset.settingForm });
            return;
        }
        if (target.dataset.settingFollow) {
            this.store.dispatch({ type: 'live-follow-mode', mode: target.dataset.settingFollow });
            return;
        }
        if (target.dataset.settingText) {
            this.store.dispatch({ type: 'live-text-mode', mode: target.dataset.settingText });
            return;
        }
        if (target.dataset.settingParticipation) {
            this.store.dispatch({ type: 'set-participation', mode: target.dataset.settingParticipation });
            return;
        }
        if (target.dataset.settingScale) {
            this.store.dispatch({ type: 'set-text-scale', value: target.dataset.settingScale });
            return;
        }
        if (target.dataset.settingMotion !== undefined) {
            this.store.dispatch({ type: 'set-reduced-motion', value: target.dataset.settingMotion === '1' });
            return;
        }
        if (target.dataset.resetPostures !== undefined) {
            this.store.dispatch({ type: 'reset-local-postures' });
            return;
        }
        const nav = target.dataset.nav;
        if (nav === 'previous')
            this.changeDate(shift(this.store.getState().selectedDate, -1));
        else if (nav === 'next')
            this.changeDate(shift(this.store.getState().selectedDate, 1));
        else if (nav === 'today')
            this.changeDate(iso(new Date()));
        const action = target.dataset.action;
        if (action) {
            this.store.dispatch({ type: 'home-action', action });
            if (action === 'follow')
                this.store.dispatch({ type: 'enter-live' });
            else if (action === 'prepare')
                this.store.dispatch({ type: 'enter-prepare' });
            else if (action === 'thanks')
                this.store.dispatch({ type: 'enter-thanksgiving' });
            else if (action === 'today-mass')
                this.store.dispatch({ type: 'home-sheet', sheet: 'mass' });
            else if (action === 'more')
                this.store.dispatch({ type: 'home-sheet', sheet: 'more' });
