function aoWhySourcesForStep(step, enrichment, lang) {
    const ids = [];
    const add = (...values) => values.forEach(v => { if (v && !ids.includes(v)) ids.push(v); });

    const id = String(step?.id || '');
    const holyWeek = /^(palms-|holy-thursday-|gf-|vigil-)/.test(id);
    if (holyWeek) add('holyweek1955','fortescue1962');
    else if (/^(candlemas-|ashes-)/.test(id)) add('fortescue1962');
    if (AO_THEOLOGY_STEPS.has(id)) add('mediatorDei','ccc');
    if (AO_HISTORY_STEPS.has(id)) add('fortescueMass','jungmann');
    (enrichment?.sources || []).forEach(add);
    return ids.map(k => {
        const base = AO_WHY_SOURCES[k];
        if (!base) return null;
        const locus = aoSourceLocus(id, k, lang);
        return locus ? { ...base, locus } : base;
    }).filter(Boolean);
}
