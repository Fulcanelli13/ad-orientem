function pbSearchNorm(v){
 return String(v??'').toLowerCase()
  .replace(/œ/g,'oe').replace(/æ/g,'ae')
  .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
  .replace(/[’'`´]/g,' ')
  .replace(/[^a-z0-9]+/g,' ')
  .trim().replace(/\s+/g,' ');
}
function pbSearchTerms(q){
 const n=pbSearchNorm(q);if(!n)return[];
 const out=new Set([n]);
 for(const group of PB_SEARCH_ALIAS_GROUPS){
  const norm=group.map(pbSearchNorm);
  if(norm.some(a=>a===n||a.includes(n)||n.includes(a)))norm.forEach(a=>out.add(a));
 }
 return [...out].filter(Boolean);
}
function pbFieldScore(value,terms,exact=120,prefix=108,contains=94){
 const v=pbSearchNorm(value);if(!v)return 0;let best=0;
 for(const q of terms){
  if(v===q)best=Math.max(best,exact);
  else if(v.startsWith(q))best=Math.max(best,prefix);
  else if(v.includes(q))best=Math.max(best,contains);
  else {const toks=q.split(' ').filter(Boolean);if(toks.length>1&&toks.every(x=>v.includes(x)))best=Math.max(best,contains-8)}
 }
 return best;
}
function pbPrayerScore(p,q){
 const terms=pbSearchTerms(q);if(!terms.length)return 1;
 const cat=D.categories[p.category]||{};
 let score=0;
 score=Math.max(score,pbFieldScore(p.title,terms,130,118,106),pbFieldScore(p.titleFr,terms,130,118,106));
 score=Math.max(score,pbFieldScore((PB_PRAYER_EXTRA_ALIASES[p.id]||[]).join(' '),terms,124,112,100));
 score=Math.max(score,pbFieldScore([p.category,cat.en,cat.fr].join(' '),terms,88,82,76));
 score=Math.max(score,pbFieldScore(p.la,terms,58,55,52),pbFieldScore(p.en,terms,56,53,50),pbFieldScore(p.fr,terms,56,53,50));
 return score;
}
function pbModuleScore(m,q){
 const terms=pbSearchTerms(q);if(!terms.length)return 1;
 let score=Math.max(pbFieldScore(m.title[0],terms,132,120,108),pbFieldScore(m.title[1],terms,132,120,108));
 score=Math.max(score,pbFieldScore((m.aliases||[]).join(' '),terms,126,114,102));
 score=Math.max(score,pbFieldScore(m.desc.join(' '),terms,68,62,58));
 return score;
}
function pbSearchPrayers(q){
 if(!pbSearchNorm(q))return Object.values(D.prayers).map(p=>({p,score:1}));
 return Object.values(D.prayers).map(p=>({p,score:pbPrayerScore(p,q)})).filter(x=>x.score>0).sort((a,b)=>b.score-a.score||prayerTitle(a.p).localeCompare(prayerTitle(b.p)));
}
function pbSearchModules(q){

}

 const startsGloria=/^(?:gloria patri|glory be|gloire au père|gloire au pere)\b/i.test(raw);

 const kind=(idPater||startsPater)?'pater':(idAve||startsAve)?'ave':(idGloria||startsGloria)?'gloria':'';
 if(!kind)return {kind:'',leader:raw,response:''};

 }[kind];

 for(const mark of markers){

  if(at>0)return {kind,leader:raw.slice(0,at).trim(),response:raw.slice(at).trim()};
 }
 return {kind,leader:raw,response:''};
}

function aoRRoleKind(id){

function aoRResponsiveText(text,id=''){

    Pater / Ave / Gloria are divided semantically by aoRPrayerHtml in Group mode,

function aoREsc(v){return String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function aoRRoleLine(line){

 if(!m)return `<span class="aoPrayerProse">${aoREsc(line)}</span>`;
 const cls=m[1]==='℣'?'aoVersicle':'aoResponse';

 return `<span class="aoPrayerDialogueLine ${cls}"><span class="aoPrayerRole" aria-label="${esc(lab)}">${m[1]}.</span><span class="aoPrayerDialogueBody"><span class="aoPrayerWords">${aoREsc(m[2])}</span></span></span>`;
}
function aoRLitanyLine(line){
 const raw=String(line||'').trim();if(!raw)return'';
 let m=raw.match(/^(.+?)\s+[—–]\s+(.+)$/);
 if(!m)m=raw.match(/^(.+?),\s*((?:ora pro nobis|miserere nobis|parce nobis(?:, Domine)?|exaudi nos(?:, Domine)?|libera nos(?:, Domine)?|pray for us|have mercy on us|spare us(?:, O Lord)?|graciously hear us(?:, O Lord)?|deliver us(?:, O Lord)?|priez pour nous|ayez pitié de nous|pardonnez-nous(?:, Seigneur)?|exaucez-nous(?:, Seigneur)?|délivrez-nous(?:, Seigneur)?)\.?)$/i);

 return `<span class="aoLitanyLine"><span class="aoLitanyCall"><span class="aoPrayerRole" aria-hidden="true">℣.</span><span>${aoREsc(m[1].trim())}</span></span><span class="aoLitanyResponse"><span class="aoPrayerRole" aria-hidden="true">℟.</span><span>${aoREsc(m[2].trim())}</span></span></span>`;
}
function aoRPrayerHtml(text,id=''){
 const shown=aoRResponsiveText(text,id),kind=aoRRoleKind(id);
 const out=[];
 let blank=false;

 for(const raw of shown.split(/\n/)){
  const line=String(raw||'').trim();
  if(!line){blank=true;continue}
  if(blank&&out.length)out.push('<span class="aoPrayerSpacer" aria-hidden="true"></span>');
  blank=false;

  if(kind==='litany'){out.push(aoRLitanyLine(line));continue}

  /* Mixed sequences such as the Angelus contain complete Ave Maria prayers
     between intrinsic versicle/response pairs. Detect the complete common prayer
     by its actual text, not only by the module identifier. */
  if(aoRMode()==='group'&&!/^[℣℟]\./.test(line)){
   const parts=aoRCommonParts(line,id);
   if(parts.response){
    out.push(`<span class="aoCustomarySplit" data-common-prayer="${esc(parts.kind)}"><span class="aoCustomaryLeader"><span class="aoSrOnly">${t('Leader','Meneur')}</span>${aoREsc(parts.leader)}</span><span class="aoCustomaryResponse"><span class="aoSrOnly">${t('Group response','Réponse du groupe')}</span>${aoREsc(parts.response)}</span></span>`);
    continue;
   }
  }

  if(/^[℣℟]\./.test(line)){out.push(aoRRoleLine(line));continue}
  out.push(`<span class="aoPrayerProse">${aoREsc(line)}</span>`);
 }
 return out.join('');
}

function aoRStripUsfm(x){
 return String(x||'')
  .replace(/\\f\s+[\s\S]*?\\f\*/g,'').replace(/\\x\s+[\s\S]*?\\x\*/g,'')
  .replace(/\\(?:add|bd|bdit|bk|dc|em|it|k|nd|ord|pn|qt|sc|sig|sls|tl|wj)\*/g,'')
  .replace(/\\(?:add|bd|bdit|bk|dc|em|it|k|nd|ord|pn|qt|sc|sig|sls|tl|wj)\s*/g,'')
  .replace(/\\[a-z0-9]+\*/gi,'').replace(/\\[a-z0-9]+\s+/gi,'').replace(/\s+/g,' ').trim();
}
function aoRParseUsfm(text,ch){
 const v={};let c=null,n=null;
 for(const raw of String(text||'').split(/\r?\n/)){
   const line=raw.trim();let m=line.match(/^\\c\s+(\d+)/);
   if(m){c=+m[1];n=null;continue} if(c!==ch)continue;
   m=line.match(/^\\v\s+(\d+)[a-z]?\s+(.*)$/i);
   if(m){n=+m[1];v[n]=aoRStripUsfm(m[2]);continue}
   if(n&&line&&!line.startsWith('\\'))v[n]+=' '+aoRStripUsfm(line);
 }
 return v;
}
